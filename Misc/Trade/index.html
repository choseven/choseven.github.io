import ccxt, time, os import numpy as np import pandas as pd from datetime import datetime API_KEY = os.environ["EXCHANGE_API_KEY"] API_SECRET = os.environ["EXCHANGE_SECRET"] SYMBOL = "BTC/USDT" TIMEFRAME = "1d" RISK = "medium" FEE_RATE = 0.001 SLIPPAGE = 0.0005 exchange = ccxt.binance({ "apiKey": API_KEY, "secret": API_SECRET, "options": {"defaultType": "spot"}, }) def fetch_ohlcv(symbol, timeframe="1d", limit=200): bars = exchange.fetch_ohlcv(symbol, timeframe, limit=limit) df = pd.DataFrame(bars, columns=["ts","open","high","low","close","vol"]) df["ts"] = pd.to_datetime(df["ts"], unit="ms") return df def ema(series, n): return series.ewm(span=n, adjust=False).mean() def rsi(series, n=14): d = series.diff() gain = d.clip(lower=0).rolling(n).mean() loss = (-d.clip(upper=0)).rolling(n).mean() rs = gain / loss.replace(0, 1e-10) return 100 - 100 / (1 + rs) def bollinger(series, n=20): m = series.rolling(n).mean() s = series.rolling(n).std() return m + 2*s, m, m - 2*s def signals(df): c = df["close"] v = df["vol"] r = rsi(c) e9, e21 = ema(c,9), ema(c,21) e50 = ema(c,50) macd_l = ema(c,12) - ema(c,26) signal_l = ema(macd_l,9) bb_u, bb_m, bb_l = bollinger(c) vol_avg = v.rolling(20).mean() roc = c.pct_change(10) * 100 atr = (df["high"] - df["low"]).rolling(14).mean() buy_score = sell_score = 0.0 reasons = [] if r.iloc[-1] < 30: buy_score += 0.85; reasons.append("RSI oversold") elif r.iloc[-1] > 70: sell_score += 0.85; reasons.append("RSI overbought") if e9.iloc[-1] > e21.iloc[-1] and e9.iloc[-2] <= e21.iloc[-2]: buy_score += 0.90; reasons.append("EMA9 golden cross") elif e9.iloc[-1] < e21.iloc[-1] and e9.iloc[-2] >= e21.iloc[-2]: sell_score += 0.90; reasons.append("EMA9 death cross") if macd_l.iloc[-1] > signal_l.iloc[-1] and macd_l.iloc[-2] <= signal_l.iloc[-2]: buy_score += 0.80; reasons.append("MACD bullish cross") elif macd_l.iloc[-1] < signal_l.iloc[-1] and macd_l.iloc[-2] >= signal_l.iloc[-2]: sell_score += 0.80; reasons.append("MACD bearish cross") if c.iloc[-1] < bb_l.iloc[-1]: buy_score += 0.70; reasons.append("Below BB") elif c.iloc[-1] > bb_u.iloc[-1]: sell_score += 0.70; reasons.append("Above BB") if v.iloc[-1] > vol_avg.iloc[-1] * 1.5: if c.iloc[-1] > c.iloc[-2]: buy_score += 0.65; reasons.append("Volume breakout") else: sell_score += 0.65; reasons.append("Volume selloff") if c.iloc[-1] > e50.iloc[-1]: buy_score += 0.55 else: sell_score += 0.55 if roc.iloc[-1] > 3: buy_score += 0.50; reasons.append("Positive ROC") elif roc.iloc[-1] < -3: sell_score += 0.50; reasons.append("Negative ROC") total = buy_score + sell_score or 1 threshold = {"low":0.60,"medium":0.52,"high":0.44}[RISK] if buy_score/total > threshold: return "BUY", buy_score/total, reasons, atr.iloc[-1] if sell_score/total > threshold: return "SELL", sell_score/total, reasons, atr.iloc[-1] return "HOLD", 0.5, reasons, atr.iloc[-1] def kelly_fraction(win_prob, win_loss_ratio=1.5): k = win_prob - (1 - win_prob) / win_loss_ratio scale = {"low":0.25,"medium":0.5,"high":0.8}[RISK] return max(0, min(0.95, k * scale)) cash = 1000.0 units = 0.0 entry_price = None wins = losses = 0 while True: try: df = fetch_ohlcv(SYMBOL, TIMEFRAME, limit=200) action, conf, reasons, atr = signals(df) price = df["close"].iloc[-1] port = cash + units * price total_trades = wins + losses win_rate = wins / total_trades if total_trades > 0 else 0.55 frac = kelly_fraction(win_rate if win_rate > 0 else 0.55) print(f"[{datetime.now():%H:%M:%S}] {action} conf={conf:.2f} price={price:.2f} port=${port:.2f}") if action == "BUY" and cash > price * 0.01: spend = cash * frac fee = spend * FEE_RATE effective_price = price * (1 + SLIPPAGE) qty = (spend - fee) / effective_price units += qty; cash -= spend; entry_price = effective_price print(f" → BUY {qty:.6f} @ {effective_price:.2f} | fee={fee:.2f}") print(f" → Reasons: {', '.join(reasons)}") elif action == "SELL" and units > 0.0001: sell_qty = units * {"low":0.5,"medium":0.75,"high":1.0}[RISK] effective_price = price * (1 - SLIPPAGE) proceeds = sell_qty * effective_price fee = proceeds * FEE_RATE pl = proceeds - fee - (sell_qty * (entry_price or price)) if pl > 0: wins += 1 else: losses += 1 units -= sell_qty; cash += proceeds - fee print(f" → SELL {sell_qty:.6f} @ {effective_price:.2f} | P&L={pl:+.2f}") if units > 0 and entry_price: stop = entry_price * (1 - {"low":0.05,"medium":0.08,"high":0.12}[RISK]) if price < stop: proceeds = units * price * (1 - FEE_RATE) cash += proceeds; losses += 1 print(f" → STOP-LOSS triggered @ {price:.2f}") units = 0; entry_price = None time.sleep(86400) except Exception as e: print(f"Error: {e}") time.sleep(60)
